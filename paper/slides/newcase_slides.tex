\begin{frame}{Setup: three held-out cases, both models frozen}
  \begin{itemize}
    \item Three newly-registered, out-of-distribution microstructures:
      \textbf{MS211}, \textbf{MS221}, \textbf{MS231} (all at $V{=}400$),
      disjoint from the frozen 16-case in-distribution set.
    \item Both models are \textbf{frozen} (no retraining): FractureTAU uses the
      base / MODE-1 fracture-mask checkpoint; ConvLSTM is Kathleen's frozen
      reference. Each rolls out \emph{independently} under the
      \textbf{identical autoregressive protocol} (exogenous channels from GT,
      fracture channel fed back, no-healing).
    \item Representative seed for FractureTAU: \textbf{seed 42}
      (\texttt{headline\_s42}); the headline aggregate below is the 3-seed mean.
    \item Headline (late-rollout macro $F_1$), cited from the provenance-tagged
      \texttt{newcase\_comparison.md} (09-08), \emph{not} re-typed here:
      FractureTAU \textbf{0.9252} vs.\ ConvLSTM \textbf{0.5974}
      --- FractureTAU wins \textbf{3/3}.
  \end{itemize}
  \vfill
  {\footnotesize Panels below align GT / FractureTAU / ConvLSTM to the same
  rollout step; the common length $L=\min$ of the three sequences (any trim is
  logged by the render script). No frames are padded or resampled.}
\end{frame}
